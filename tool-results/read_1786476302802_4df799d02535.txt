   120→   120→function loadHistory(): GeneratedImage[] {
   121→   121→  if (typeof window === "undefined") return [];
   122→   122→  try {
   123→   123→    const raw = window.localStorage.getItem(HISTORY_KEY);
   124→   124→    if (!raw) return [];
   125→   125→    const parsed = JSON.parse(raw);
   126→   126→    if (!Array.isArray(parsed)) return [];
   127→   127→    return parsed as GeneratedImage[];
   128→   128→  } catch {
   129→   129→    return [];
   130→   130→  }
   131→   131→}
   132→   132→
   133→   133→function saveHistory(items: GeneratedImage[]) {
   134→   134→  if (typeof window === "undefined") return;
   135→   135→  try {
   136→   136→    // Cap to MAX_HISTORY to avoid blowing the localStorage quota.
   137→   137→    // (Each PNG data-uri can be ~1-2 MB; with 50 items we may still hit quota —
   138→   138→    //  in that case we drop the oldest until it fits.)
   139→   139→    let capped = items.slice(0, MAX_HISTORY);
   140→   140→    while (capped.length > 0) {
   141→   141→      try {
   142→   142→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(capped));
   143→   143→        return;
   144→   144→      } catch {
   145→   145→        // Quota exceeded — drop the oldest and retry.
   146→   146→        capped = capped.slice(0, capped.length - 1);
   147→   147→      }
   148→   148→    }
   149→   149→  } catch {
   150→   150→    /* noop */
   151→   151→  }
   152→   152→}
   153→   153→
   154→   154→function downloadDataUri(dataUri: string, filename: string) {
   155→   155→  const a = document.createElement("a");
   156→   156→  a.href = dataUri;
   157→   157→  a.download = filename;
   158→   158→  document.body.appendChild(a);
   159→   159→  a.click();
   160→   160→  document.body.removeChild(a);
   161→   161→}

   163→   163→function formatBytes(b: number): string {
   164→   164→  if (b < 1024) return `${b} B`;
   165→   165→  if (b < 1024 * 1024) return `${(b / 1024).toFixed(1)} KB`;
   166→   166→  return `${(b / (1024 * 1024)).toFixed(2)} MB`;
   167→   167→}

   169→   169→function aspectRatioClass(size: ImageSize): string {
   170→   170→  switch (size) {
   171→   171→    case "1024x1024": return "aspect-square";
   172→   172→    case "768x1344":
   173→   173→    case "720x1440": return "aspect-[9/16]";
   174→   174→    case "864x1152": return "aspect-[3/4]";
   175→   175→    case "1344x768":
   176→   176→    case "1440x720": return "aspect-video";
   177→   177→    case "1152x864": return "aspect-[4/3]";
   178→   178→    default: return "aspect-square";
   179→   179→  }
   180→   180→}
