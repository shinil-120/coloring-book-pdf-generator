   113→const HISTORY_KEY = "zai-image-gen-history-v1";
   114→const MAX_HISTORY = 50;

   116→// ─────────────────────────────────────────────────────────────────────────
   117→// Helpers
   118→// ─────────────────────────────────────────────────────────────────────────
   119→
   120→function loadHistory(): GeneratedImage[] {
   121→  if (typeof window === "undefined") return [];
   122→  try {
   123→    const raw = window.localStorage.getItem(HISTORY_KEY);
   124→    if (!raw) return [];
   125→    const parsed = JSON.parse(raw);
   126→    if (!Array.isArray(parsed)) return [];
   127→    return parsed as GeneratedImage[];
   128→  } catch {
   129→    return [];
   130→  }
   131→}
   132→
   133→function saveHistory(items: GeneratedImage[]) {
   134→  if (typeof window === "undefined") return;
   135→  try {
   136→    // Cap to MAX_HISTORY to avoid blowing the localStorage quota.
   137→    // (Each PNG data-uri can be ~1-2 MB; with 50 items we may still hit quota —
   138→    //  in that case we drop the oldest until it fits.)
   139→    let capped = items.slice(0, MAX_HISTORY);
   140→    while (capped.length > 0) {
   141→      try {
   142→        window.localStorage.setItem(HISTORY_KEY, JSON.stringify(capped));
   143→        return;
   144→      } catch {
   145→        // Quota exceeded — drop the oldest and retry.
   146→        capped = capped.slice(0, capped.length - 1);
   147→      }
   148→    }
   149→  } catch {
   150→    /* noop */
   151→  }
   152→}
   153→
   154→function downloadDataUri(dataUri: string, filename: string) {
   155→  const a = document.createElement("a");
   156→  a.href = dataUri;
   157→  a.download = filename;
   158→  document.body.appendChild(a);
   159→  a.click();
   160→  document.body.removeChild(a);
   161→}
   162→
   163→function formatBytes(b: number): string {
   164→  if (b < 1024) return `${b} B`;
   165→  if (b < 1024 * 1024) return `${(b / 1024).toFixed(1)} KB`;
   166→  return `${(b / (1024 * 1024)).toFixed(2)} MB`;
   167→}
   168→
   169→function aspectRatioClass(size: ImageSize): string {
   170→  switch (size) {
   171→    case "1024x1024": return "aspect-square";
   172→    case "768x1344":
   173→    case "720x1440": return "aspect-[9/16]";
   174→    case "864x1152": return "aspect-[3/4]";
   175→    case "1344x768":
   176→    case "1440x720": return "aspect-video";
   177→    case "1152x864": return "aspect-[4/3]";
   178→    default: return "aspect-square";
   179→  }
   180→}

   220→      toast.error("Please enter a prompt first");

   224→    // Abort any previous in-flight request
   225→    if (abortRef.current) abortRef.current.abort();

   229→    setLoading(true);

   233→      const res = await fetch("/api/generate-image", {
   234→        method: "POST",

   237→          prompt: trimmed,
   238→          size,
   239→          style,

   243→        signal: controller.signal,

   257→        styleLabel: data.styleLabel,
   258→        seed: data.seed,
   259→        bytes: data.bytes,
   260→        generatedAt: data.generatedAt,

   264→      setHistory((prev) => [item, ...prev].slice(0, MAX_HISTORY));

   270→      const msg = err instanceof Error ? err.message : String(err);
   271→      setError(msg);
   272→      toast.error("Generation failed", { description: msg.slice(0, 200) });

   275→      abortRef.current = null;
